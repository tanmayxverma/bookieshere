CREATE OR REPLACE FUNCTION public.has_role(_user_id UUID, _role app_role)
RETURNS BOOLEAN
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path = public
AS $$

    FROM public.user_roles
    WHERE user_id = _user_id
      AND role = _role
  )

-- Create a function to clean up stale sessions (no heartbeat for 60 seconds)
CREATE OR REPLACE FUNCTION public.cleanup_stale_sessions()
RETURNS void
LANGUAGE sql
SECURITY DEFINER
SET search_path = public
AS $$
  UPDATE public.study_sessions
  SET status = 'completed',
      ended_at = last_heartbeat,
      duration_seconds = EXTRACT(EPOCH FROM (last_heartbeat - started_at))::integer
  WHERE status = 'active'
    AND last_heartbeat < now() - interval '60 seconds';
$$;

RETURNS SETOF public.profiles

CREATE OR REPLACE FUNCTION public.upsert_guest_profile(
  _device_id text,
  _display_name text,
  _target_category text,
  _avatar_url text DEFAULT NULL
)
RETURNS public.profiles
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  result public.profiles;
BEGIN
  IF _device_id IS NULL OR length(_device_id) < 8 THEN
    RAISE EXCEPTION 'invalid device id';
  END IF;

  INSERT INTO public.profiles (device_id, account_type, display_name, avatar_url, target_category, onboarded_at)
  VALUES (_device_id, 'guest', COALESCE(NULLIF(trim(_display_name), ''), 'Aspirant'),
          _avatar_url, COALESCE(NULLIF(trim(_target_category), ''), 'JEE 2027'), now())
  ON CONFLICT (device_id) DO UPDATE
    SET display_name = EXCLUDED.display_name,
        avatar_url = EXCLUDED.avatar_url,
        target_category = EXCLUDED.target_category,
        onboarded_at = COALESCE(public.profiles.onboarded_at, now()),
        updated_at = now()
  RETURNING * INTO result;
  RETURN result;
END;
$$;

GRANT EXECUTE ON FUNCTION public.get_guest_profile(text) TO anon, authenticated;
GRANT EXECUTE ON FUNCTION public.upsert_guest_profile(text, text, text, text) TO anon, authenticated;

-- ============ auto-create profile on signup ============
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  meta jsonb := COALESCE(NEW.raw_user_meta_data, '{}'::jsonb);
  is_google boolean := COALESCE(NEW.raw_app_meta_data->>'provider', '') = 'google';
  display text := COALESCE(
    meta->>'full_name',
    meta->>'name',
    split_part(NEW.email, '@', 1),
    'Aspirant'
  );
  avatar text := COALESCE(meta->>'avatar_url', meta->>'picture');
BEGIN
  INSERT INTO public.profiles (user_id, account_type, display_name, avatar_url)
  VALUES (
    NEW.id,
    CASE WHEN is_google THEN 'google' ELSE 'email' END,
    display,
    avatar
  )
  ON CONFLICT (user_id) DO NOTHING;
  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users;
CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();

STABLE SECURITY DEFINER

SECURITY DEFINER
SET search_path = public, extensions
AS $$
DECLARE
  _row public.ypt_rooms;
BEGIN
  UPDATE public.ypt_rooms
    SET is_locked = _is_locked,
        pin_hash = CASE
          WHEN _pin IS NULL OR length(_pin) = 0 THEN NULL
          ELSE encode(extensions.digest(_pin, 'sha256'), 'hex')
        END,
        max_participants = _max
    WHERE id = _room_id AND created_by = _creator AND is_permanent = false
    RETURNING * INTO _row;
  IF _row.id IS NULL THEN RAISE EXCEPTION 'not authorized or room not found'; END IF;
  RETURN _row;
END;
$$;

  IF _pid IS NULL THEN RAISE EXCEPTION 'profile not found'; END IF;
  SELECT * INTO _row FROM public.support_chats WHERE profile_id = _pid;
  IF _row.id IS NULL THEN INSERT INTO public.support_chats(profile_id) VALUES(_pid) RETURNING * INTO _row; END IF;
  RETURN _row;
END $$;

CREATE OR REPLACE FUNCTION public.guest_send_support_message(_device_id text, _body text, _image_url text DEFAULT NULL)
RETURNS public.support_messages LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
DECLARE _pid uuid; _cid uuid; _status text; _row public.support_messages;
BEGIN
  SELECT id INTO _pid FROM public.profiles WHERE device_id = _device_id AND user_id IS NULL LIMIT 1;
  IF _pid IS NULL THEN RAISE EXCEPTION 'guest profile not found'; END IF;
  SELECT id, status INTO _cid, _status FROM public.support_chats WHERE profile_id = _pid;
  IF _cid IS NULL THEN
    INSERT INTO public.support_chats(profile_id) VALUES(_pid) RETURNING id, status INTO _cid, _status;
  END IF;
  IF _status <> 'accepted' THEN RAISE EXCEPTION 'chat not accepted yet'; END IF;
  INSERT INTO public.support_messages(chat_id, author_role, body, image_url)
    VALUES (_cid, 'user', _body, _image_url) RETURNING * INTO _row;
  RETURN _row;
END $$;

CREATE OR REPLACE FUNCTION public.guest_list_support_messages(_device_id text)
RETURNS SETOF public.support_messages LANGUAGE sql STABLE SECURITY DEFINER SET search_path = public AS $$
  SELECT m.* FROM public.support_messages m
  JOIN public.support_chats c ON c.id = m.chat_id
  JOIN public.profiles p ON p.id = c.profile_id
  WHERE p.device_id = _device_id AND p.user_id IS NULL
  ORDER BY m.created_at;
$$;

CREATE OR REPLACE FUNCTION public.guest_get_support_chat(_device_id text)

  SELECT c.* FROM public.support_chats c
  JOIN public.profiles p ON p.id = c.profile_id
  WHERE p.device_id = _device_id AND p.user_id IS NULL;
$$;

-- Migrate old tickets into one chat per profile
DO $$
DECLARE _p record; _chat_id uuid;
BEGIN
  FOR _p IN SELECT DISTINCT profile_id FROM public.tickets LOOP
    INSERT INTO public.support_chats(profile_id, status)
      VALUES (_p.profile_id, 'accepted')
      ON CONFLICT (profile_id) DO UPDATE SET status = 'accepted'
      RETURNING id INTO _chat_id;
    INSERT INTO public.support_messages(chat_id, author_role, body, created_at)
    SELECT _chat_id, 'system', '#'||substr(t.id::text,1,8)||' â€” '||t.subject, t.created_at
    FROM public.tickets t WHERE t.profile_id = _p.profile_id;
    INSERT INTO public.support_messages(chat_id, author_role, body, created_at)
    SELECT _chat_id, m.author_role, m.body, m.created_at
    FROM public.ticket_messages m

  WITH CHECK (

CREATE OR REPLACE FUNCTION public.is_user_banned(_user_id uuid)
RETURNS boolean LANGUAGE sql STABLE SECURITY DEFINER SET search_path = public AS $$
  SELECT COALESCE((SELECT is_banned FROM public.profiles WHERE user_id = _user_id LIMIT 1), false);
$$;

CREATE OR REPLACE FUNCTION public.is_user_muted(_user_id uuid)
RETURNS boolean LANGUAGE sql STABLE SECURITY DEFINER SET search_path = public AS $$
  SELECT COALESCE((
    SELECT is_banned OR (is_muted AND (muted_until IS NULL OR muted_until > now()))
    FROM public.profiles WHERE user_id = _user_id LIMIT 1
  ), false);
$$;

CREATE OR REPLACE FUNCTION public.block_muted_authors()
RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
BEGIN
  IF public.is_user_muted(NEW.user_id) THEN
    RAISE EXCEPTION 'You are currently restricted from posting in the community.';
  END IF;
  RETURN NEW;
END $$;

CREATE OR REPLACE FUNCTION public.block_banned_authors()
RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
BEGIN
  IF public.is_user_banned(NEW.user_id) THEN
    RAISE EXCEPTION 'Your account is banned from the community.';
  END IF;
  RETURN NEW;
END $$;

CREATE OR REPLACE FUNCTION public.support_reopen_on_user_message()
RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
BEGIN
  IF NEW.author_role = 'user' THEN
    UPDATE public.support_chats
      SET status = 'pending', updated_at = now()
      WHERE id = NEW.chat_id AND status <> 'pending';
  END IF;
  RETURN NEW;
END $$;
